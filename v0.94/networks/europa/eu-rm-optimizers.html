    RM Optimizers¶

    This vignette examines swapping out optimizers for carriers in the Europa network.

    import passengersim as pax

pax.versions()

    passengersim 0.94
passengersim.core 0.94

    cfg = pax.Config.from_yaml(pax.demo_network("europa"))

    ex = pax.Experiments(cfg, output_dir=False)

    @ex
def EEE(cfg: pax.Config):
    return cfg

    Setup ProBP Experiments¶

    @ex
def PEE(cfg: pax.Config):
    cfg.carriers["Brat"].rm_system = "P"
    return cfg


@ex
def EPE(cfg: pax.Config):
    cfg.carriers["Stroop"].rm_system = "P"
    return cfg


@ex
def EEP(cfg: pax.Config):
    cfg.carriers["Tapas"].rm_system = "P"
    return cfg


@ex
def PPE(cfg: pax.Config):
    cfg.carriers["Brat"].rm_system = "P"
    cfg.carriers["Stroop"].rm_system = "P"
    return cfg


@ex
def PPP(cfg: pax.Config):
    cfg.carriers["Brat"].rm_system = "P"
    cfg.carriers["Stroop"].rm_system = "P"
    cfg.carriers["Tapas"].rm_system = "P"
    return cfg
